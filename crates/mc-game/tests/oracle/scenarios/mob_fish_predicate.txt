# `execute if predicate` tags the fish whose own value matches, and a different value must not match.

level flat

entity_counter 8200

spawn_player Steve creative 0.5 -60 0.5 0 0
mobspawn SAL minecraft:salmon 3.5 -60 4.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b,Tags:["sal"],type:"large"}
