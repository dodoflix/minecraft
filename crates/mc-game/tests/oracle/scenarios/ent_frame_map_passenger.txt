# An item frame that gains a passenger in a tick when it updates its map: sendChanges sends the set-passengers packet first, then the map
# updates of the frame (every player of the level ticks the map), then the frame's own changed data. The watchers get the packets in that order.
command_env
level flat
preload 4
level_seed 5
entity_counter 900
spawn_player A survival 5.5 -60 2.5 90 0
spawn_player B survival 24.5 -60 2.5 90 0
block 3 -60 0 minecraft:stone
tick 2
dump packets:A:none packets:B:none
note the first map of the world goes to A
give A hotbar0 minecraft:map 2
select A 0
use A main
tick 1
note a frame with the map and a cow riding it appear in the same tick: the frame sends its changes in its first tick
spawn f1 minecraft:item_frame 4.5 -59.5 0.5 0 0 {Facing:5b,Tags:[one],Item:{id:"minecraft:filled_map",count:1,components:{"minecraft:map_id":0}}}
entity_seed f1 21
mobspawn cow minecraft:cow 4.5 -59.5 0.5 0 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed cow 22
ride cow f1
watch packets:A:map_item_data,set_passengers,set_entity_data,add_entity packets:B:map_item_data,set_passengers,set_entity_data,add_entity
tick 12
watch off
tick 12
dump xentities
