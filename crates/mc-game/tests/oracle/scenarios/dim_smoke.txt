# the replay holds the three levels of a server: they tick together, `in DIM` and `execute in` act on one of them, a player and a mob
# travel between them (`tp_dim`, `/tp`). The nether and the end are only touched high above their terrain (y 200).
level void
command_env
levels overworld,the_nether,the_end
preload 2
difficulty peaceful
entity_counter 100
dump levels
in the_nether block 0 200 0 minecraft:netherrack
in the_end block 0 200 0 minecraft:end_stone
cmd execute in minecraft:the_nether run setblock 1 200 0 minecraft:stone
cmd execute in minecraft:the_end run setblock 1 200 0 minecraft:obsidian
cmd execute if dimension minecraft:overworld
cmd execute if dimension minecraft:the_nether
cmd execute in minecraft:the_nether if dimension minecraft:the_nether
tick 2
note blocks of the three levels
in the_nether dump blocks:0,200,0,1,200,0
in the_end dump blocks:0,200,0,1,200,0
dump blocks:0,200,0,1,200,0
spawn_player P creative 0.5 -60 0.5 0 0
spawn Z minecraft:pig 3.5 -60 0.5 0 0 {NoAI:1b,NoGravity:1b,PersistenceRequired:1b}
entity_seed Z 7
tick 2
dump packets:P:none levels players mob:Z
note the player enters the nether
tp_dim P the_nether 0.5 201 0.5
dump packets:P:respawn,player_position,change_difficulty,player_abilities,set_experience,set_held_slot,update_mob_effect
tick 2
dump levels players player:P
note the mob enters the end
tp_dim Z the_end 0.5 201 0.5
tick 2
dump levels
in the_end dump entity:Z
note a command moves the player inside the nether, then to the end
cmd execute in minecraft:the_nether run tp P 4.5 202 0.5
cmd execute in minecraft:the_end run tp P 0.5 201 0.5
tick 2
dump levels players packets:P:respawn,player_position
note a command moves the mob back to the overworld
cmd execute in minecraft:overworld run tp @e[type=minecraft:pig,limit=1] 0.5 -60 0.5
tick 2
dump levels
dump mob:pig#1 players
