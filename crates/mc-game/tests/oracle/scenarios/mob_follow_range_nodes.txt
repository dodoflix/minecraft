# A new follow range resizes the node budget of the mob's path search (Mob.onAttributeUpdated): a zombie behind an 23 block wall needs
# more than the 560 nodes of a follow range of 35 to find the way round, so the path is only partial until /attribute raises the range
# and partial again once it is lowered.
command_env
level flat
preload 6
difficulty normal
entity_counter 9700
level_seed 9701
fill -11 -60 6 11 -58 6 minecraft:stone
mobspawn Z minecraft:zombie 0.5 -60 0.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
tick 2
set Z on_ground true
path_query Z 0.5 -60 12.5
cmd attribute @e[type=zombie,limit=1] minecraft:follow_range base set 40
path_query Z 0.5 -60 12.5
cmd attribute @e[type=zombie,limit=1] minecraft:follow_range base set 50
path_query Z 0.5 -60 12.5
cmd attribute @e[type=zombie,limit=1] minecraft:follow_range base set 35
path_query Z 0.5 -60 12.5
