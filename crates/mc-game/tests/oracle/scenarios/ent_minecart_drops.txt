# Destroying a minecart: the item drop carries the custom name, the entity_drops rule turns it off, a creative player discards without a drop,
# a rider is ejected when the cart breaks.
level flat
preload 3
level_seed 6
entity_counter 810
fill -2 -60 -2 40 -40 12 minecraft:air
spawn_player Alice survival 20.5 -60 5.5 0 0
spawn_player Bob creative 24.5 -60 5.5 0 0
spawn c1 minecraft:minecart 0.5 -59.9375 0.5 0 0 {CustomName:'"Rusty"'}
entity_seed c1 21
spawn c2 minecraft:minecart 3.5 -59.9375 0.5 0 0 {}
entity_seed c2 22
spawn c3 minecraft:minecart 6.5 -59.9375 0.5 0 0 {CustomName:'"Shiny"'}
entity_seed c3 23
spawn c4 minecraft:minecart 9.5 -59.9375 0.5 0 0 {}
entity_seed c4 24
spawn v1 minecraft:pig 9.5 -59.9375 0.5 0 0 {NoAI:1b}
entity_seed v1 25
ride v1 c4
tick 2
dump packets:Alice:none packets:Bob:none
note a named cart broken by a mob: the drop keeps the name
hurt c1 mob_attack 50
tick 3
dump xentities packets:Alice:add_entity,remove_entities,set_entity_data
note a cart broken in two hits by a player
hurt c2 player_attack 3 by=Alice direct=Alice
tick 2
dump xentities
hurt c2 player_attack 2 by=Alice direct=Alice
tick 3
dump xentities packets:Alice:add_entity,remove_entities,set_entity_data
note a creative player discards the named cart: no drop
hurt c3 player_attack 1 by=Bob direct=Bob
tick 3
dump xentities packets:Alice:add_entity,remove_entities
note a cart with a rider breaks: the rider stays
hurt c4 mob_attack 50
tick 3
dump xentities packets:Alice:add_entity,remove_entities,set_passengers
note entity_drops off: nothing drops
gamerule entity_drops false
spawn c5 minecraft:minecart 12.5 -59.9375 0.5 0 0 {CustomName:'"Dull"'}
entity_seed c5 26
tick 1
hurt c5 mob_attack 50
tick 3
dump xentities packets:Alice:add_entity,remove_entities
