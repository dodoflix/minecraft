# The scale attribute set with /attribute on ten mobs of different classes (a cow, a baby sheep, a zombie, a baby husk, a villager, a wolf,
# a creeper, a skeleton, a pig, a horse): the size, eye height and bounding box follow getScale() whatever class overrides the default
# dimensions, a baby multiplies its age scale, and the riders of a horse, a pig and a zombie sit at the attachment point of the vehicle at the
# scale of the vehicle.
command_env
level flat
difficulty normal
preload 8
entity_counter 2300
level_seed 2301
spawn_player Steve survival 100.5 -60 100.5 0 0
mobspawn Cow minecraft:cow 4.5 -60 4.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
mobspawn Sheep minecraft:sheep 16.5 -60 4.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b,Age:-24000}
mobspawn Zombie minecraft:zombie 28.5 -60 4.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
