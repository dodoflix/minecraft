# Scoreboard objectives whose criterion is a statistic follow the statistic of the player: a custom statistic (jumps, distance walked,
# deaths) and the item statistics (mined, used, broken, picked_up, dropped, crafted, killed, killed_by). The scoreboard is dumped after every group.
command_env
level flat
preload 3
difficulty normal
entity_counter 51500
level_seed 51501
spawn_player A survival 0.5 -60 0.5 0 0
block 4 -60 0 minecraft:stone
spawn husk1 minecraft:husk 8.5 -60 0.5 180 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed husk1 51502
spawn husk2 minecraft:husk 12.5 -60 0.5 180 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed husk2 51503
tick 25
cmd scoreboard objectives add jumps minecraft.custom:minecraft.jump
cmd scoreboard objectives add walked minecraft.custom:minecraft.walk_one_cm
cmd scoreboard objectives add mined minecraft.mined:minecraft.stone
cmd scoreboard objectives add used minecraft.used:minecraft.iron_pickaxe
cmd scoreboard objectives add broken minecraft.broken:minecraft.iron_pickaxe
cmd scoreboard objectives add picked minecraft.picked_up:minecraft.cobblestone
cmd scoreboard objectives add dropped minecraft.dropped:minecraft.cobblestone
cmd scoreboard objectives add crafted minecraft.crafted:minecraft.oak_planks
cmd scoreboard objectives add killed minecraft.killed:minecraft.husk
cmd scoreboard objectives add killedby minecraft.killed_by:minecraft.husk
cmd scoreboard objectives add deaths minecraft.custom:minecraft.deaths
dump scoreboard
note one block walked, then a jump
move_packet A 1.5 -60 0.5 true
client_tick_end A
tick 1
move_packet A 2.5 -59.2 0.5 false
client_tick_end A
tick 1
move_packet A 2.5 -60 0.5 true
client_tick_end A
tick 1
dump scoreboard stats:A
note mining a stone block with a pickaxe that has one use left, then picking up the cobblestone and dropping it
give A mainhand minecraft:iron_pickaxe[damage=249] 1
snap A 3.5 -60 0.5
tick 1
dig A 4 -60 0 60
tick 20
dump scoreboard stats:A inventory:A
level_seed 51504
drop A all
tick 2
dump scoreboard stats:A
note crafting planks from a log in the inventory grid
give A hotbar4 minecraft:oak_log 1
select A 4
click A 40 0 PICKUP
click A 1 0 PICKUP
click A 0 0 QUICK_MOVE
tick 1
dump scoreboard stats:A
note killing a husk, then being killed by one
select A 0
give A mainhand minecraft:diamond_sword 1
snap A 8.5 -60 -1.5 0 0
tick 25
set husk1 health 1
attack A husk1
tick 2
dump scoreboard stats:A
hurt A minecraft:mob_attack 100 by=husk2 direct=husk2
tick 1
dump scoreboard stats:A
