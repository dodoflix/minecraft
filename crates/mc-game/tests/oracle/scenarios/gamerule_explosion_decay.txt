# The drop decay of an explosion by its source: tnt_explosion_drop_decay (off by default) and mob_explosion_drop_decay (on by default),
# each switched, with the level random fixed so the survivors are the same.
level flat
preload 3
level_seed 321
fill 0 -60 0 8 -58 8 minecraft:dirt
fill 0 -60 12 8 -58 12 minecraft:dirt
fill 0 -60 16 8 -58 16 minecraft:dirt
fill 0 -60 20 8 -58 20 minecraft:dirt
note tnt decay off (the default)
explode 4 -57 4 3 tnt
dump blocks:-1,-61,-1,9,-57,9 events entities
tick 2
dump entities
note tnt decay on
gamerule tnt_explosion_drop_decay true
explode 4 -57 12 3 tnt
dump blocks:-1,-61,11,9,-57,13 events entities
tick 2
dump entities
note mob decay on (the default)
explode 4 -57 16 3 mob
dump blocks:-1,-61,15,9,-57,17 events entities
tick 2
dump entities
note mob decay off
gamerule mob_explosion_drop_decay false
explode 4 -57 20 3 mob
dump blocks:-1,-61,19,9,-57,21 events entities
tick 2
dump entities
