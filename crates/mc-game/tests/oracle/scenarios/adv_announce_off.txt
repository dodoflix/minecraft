# The announce_advancements rule: with show_advancement_messages off a completed advancement is sent in update_advancements but not
# announced in chat; with it on it is announced to every player.
command_env
level flat
preload 2
level_seed 5
entity_counter 30
spawn_player Steve survival 0.5 -60 0.5 0 0
spawn_player Alex survival 4.5 -60 0.5 0 0
tick 2
dump packets:Steve:none packets:Alex:none
cmd_as -4 gamerule show_advancement_messages false
cmd_as -4 advancement grant Steve only minecraft:story/root
cmd_as -4 advancement grant Steve only minecraft:story/mine_stone
tick 1
dump advpackets:Steve:update_advancements,system_chat advpackets:Alex:update_advancements,system_chat advancements:Steve
cmd_as -4 gamerule show_advancement_messages true
cmd_as -4 advancement grant Steve only minecraft:story/upgrade_tools
cmd_as -4 advancement grant Alex only minecraft:story/root
cmd_as -4 advancement grant Alex only minecraft:story/mine_stone
tick 1
dump advpackets:Steve:update_advancements,system_chat advpackets:Alex:update_advancements,system_chat advancements:Steve advancements:Alex
