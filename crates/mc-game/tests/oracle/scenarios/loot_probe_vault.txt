# Vault rewards read through a data pack table (`loot_sites`) that drops a stick for the player as `this_entity`, dirt for the trial key as the tool,
# coal for the origin and one emerald per point of luck: blk_vault_unlock with the Luck III effect.
datapack loot_sites
command_env
level flat
time 6000
spawn_player p survival 8.5 -60 9.5 0 0
tick 10
effect p minecraft:luck 2 100000
block 8 -60 12 minecraft:vault[facing=north]
dump be:8,-60,12 blocks:8,-60,12,8,-60,12
watch events
tick 5
dump be:8,-60,12 blocks:8,-60,12,8,-60,12
give p mainhand minecraft:stick
use_on p 8 -60 12 north main
tick 2
give p mainhand minecraft:ominous_trial_key
use_on p 8 -60 12 north main
tick 2
dump inventory:p
give p mainhand minecraft:trial_key
entity_counter 100
level_seed 31
use_on p 8 -60 12 north main
dump inventory:p be:8,-60,12 blocks:8,-60,12,8,-60,12
