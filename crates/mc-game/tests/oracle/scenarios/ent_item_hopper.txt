# A hopper sucks in an item entity: all of it (the entity is discarded), part of it (the entity keeps the rest and tells the clients), none of
# it (a hopper without room for that item), and an item the hopper takes over two cooldowns.
command_env
level flat
preload 3
level_seed 5
entity_counter 800
spawn_player A survival 0.5 -60 20.5 0 0
tick 2
dump packets:A:none
block 2 -60 2 minecraft:hopper[facing=north,enabled=true]
be_nbt 2 -60 2 {Items:[{Slot:0b,id:"minecraft:cobblestone",count:64},{Slot:1b,id:"minecraft:cobblestone",count:64},{Slot:2b,id:"minecraft:cobblestone",count:64},{Slot:3b,id:"minecraft:cobblestone",count:64},{Slot:4b,id:"minecraft:dirt",count:60}]}
block 6 -60 2 minecraft:hopper[facing=north,enabled=true]
be_nbt 6 -60 2 {Items:[{Slot:0b,id:"minecraft:cobblestone",count:30}]}
tick 1
note part of a stack fits, none of another
spawn_item p1 minecraft:dirt 2.5 -59.5 2.5 0 0 0 10
entity_seed p1 11
spawn_item p2 minecraft:gold_ingot 2.5 -59.2 2.5 0 0 0 5
entity_seed p2 12
note all of it, then a second stack that fills the last free slot
spawn_item p3 minecraft:cobblestone 6.5 -59.5 6.5 0 0 0 4
entity_seed p3 13
spawn_item p4 minecraft:cobblestone 6.5 -59.5 2.5 0 0 0 7
entity_seed p4 14
spawn_item p5 minecraft:diamond 6.5 -59.2 2.5 0 0 0 7
entity_seed p5 15
dump xentities be:2,-60,2 be:6,-60,2 packets:A:set_entity_data,remove_entities
tick 1
dump xentities be:2,-60,2 be:6,-60,2 packets:A:set_entity_data,remove_entities
tick 8
dump xentities be:2,-60,2 be:6,-60,2 packets:A:set_entity_data,remove_entities
tick 8
dump xentities be:2,-60,2 be:6,-60,2 packets:A:set_entity_data,remove_entities
