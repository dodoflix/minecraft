# 3x3 recipes from the recipe book placed into the 2x2 grid of the inventory menu: the grid slots, the result slot and the packets that vanilla produces for a recipe wider than the grid.
command_env
level flat
preload 4
spawn_player C survival 15.5 -60 7.5 0 0
give C inv9 minecraft:oak_planks 64
give C inv10 minecraft:cobblestone 64
tick 2
cmd recipe give C minecraft:chest
cmd recipe give C minecraft:furnace
dump packets:C:none
place_recipe C minecraft:chest
dump menu:C inventory:C packets:C:container_set_slot,container_set_content,set_cursor_item,place_ghost_recipe
place_recipe C minecraft:furnace all
dump menu:C inventory:C packets:C:container_set_slot,container_set_content,set_cursor_item,place_ghost_recipe
