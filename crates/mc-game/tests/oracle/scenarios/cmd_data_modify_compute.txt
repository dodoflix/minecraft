# `data modify ... compute default | block | entity float | integer`: the provider is evaluated without error handling (an arithmetic
# failure or a non-finite float stores 0) and the result is stored as an int or float tag through every modification mode.
command_env
level flat
preload 2
level_seed 3
entity_counter 50
spawn_player A survival 0.5 -60 0.5 0 0
tick 1
cmd scoreboard objectives add smoke dummy
cmd scoreboard players set A smoke 7
cmd summon minecraft:pig 8 -60 8 {NoAI:1b,Age:-24000}
cmd setblock 2 -60 2 minecraft:smoker
cmd data merge storage smoke:r {list:[1,2],old:5,text:"x"}
note storage targets
cmd data modify storage smoke:r a set compute default integer {type:"minecraft:constant",value:5}
cmd data modify storage smoke:r b set compute default float {type:"minecraft:constant",value:2.5}
cmd data modify storage smoke:r c set compute default float {type:"minecraft:constant",value:4}
cmd data modify storage smoke:r d set compute default integer {type:"minecraft:add",inputs:[10,20]}
cmd data modify storage smoke:r old set compute default integer {type:"minecraft:constant",value:6}
cmd data modify storage smoke:r list append compute default integer {type:"minecraft:constant",value:3}
cmd data modify storage smoke:r list prepend compute default float {type:"minecraft:constant",value:0.5}
cmd data get storage smoke:r list
cmd data modify storage smoke:r list insert 1 compute default integer {type:"minecraft:constant",value:9}
cmd data modify storage smoke:r list[0] set compute default integer {type:"minecraft:constant",value:-1}
cmd data get storage smoke:r list
cmd data modify storage smoke:r text set compute default integer {type:"minecraft:constant",value:1}
cmd data modify storage smoke:r nested.deep set compute default float {type:"minecraft:constant",value:-0.25}
cmd data modify storage smoke:r merge compute default integer {type:"minecraft:constant",value:1}
cmd data modify storage smoke:r list set compute default integer {type:"minecraft:constant",value:1}
