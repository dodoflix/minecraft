# A smelting recipe from the recipe book placed at a crafting table: the recipe is of another class than the menu's, vanilla's cast throws
# inside the packet handler (the packet processor suppresses it), so nothing moves and nothing is sent.
command_env
level flat
preload 4
spawn_player A survival 3.5 -60 7.5 0 0
block 3 -60 4 minecraft:crafting_table
give A inv9 minecraft:raw_iron 8
give A inv10 minecraft:oak_planks 64
tick 2
cmd recipe give A minecraft:iron_ingot_from_smelting_raw_iron
dump packets:A:none
use_on A 3 -60 4 up main
dump menu:A packets:A:open_screen,container_set_content
place_recipe A minecraft:iron_ingot_from_smelting_raw_iron
dump menu:A inventory:A packets:A:container_set_slot,container_set_content,set_cursor_item,place_ghost_recipe
place_recipe A minecraft:iron_ingot_from_smelting_raw_iron all
dump menu:A inventory:A packets:A:container_set_slot,container_set_content,set_cursor_item,place_ghost_recipe
