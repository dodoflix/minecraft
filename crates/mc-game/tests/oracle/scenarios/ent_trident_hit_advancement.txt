# The projectile that hurts a mob is the direct entity of the damage the player_hurt_entity trigger sees: a thrown trident
# completes adventure/throw_trident, an arrow owned by the player completes adventure/shoot_arrow, an arrow of nobody completes
# nothing.
command_env
level flat
preload 3
difficulty normal
level_seed 5
entity_counter 1500
spawn_player A survival 0.5 -60 0.5 0 0
mobspawn cow1 minecraft:cow 0.5 -60 8.5 0 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed cow1 1
mobspawn cow2 minecraft:cow 6.5 -60 8.5 0 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed cow2 2
mobspawn cow3 minecraft:cow 12.5 -60 8.5 0 0 {NoAI:1b,PersistenceRequired:1b}
entity_seed cow3 3
give A hotbar0 minecraft:trident
tick 2
dump packets:A:none advancements:A
note the trident thrown straight ahead hits the cow
use A main 1
tick 10
level_seed 101
release A 1
tick 6
dump entities advancements:A stats:A packets:A:damage_event,hurt_animation,add_entity,remove_entities
note an arrow shot by the player hits the second cow
spawn arrow1 minecraft:arrow 6.5 -59.5 2.5 0 0 {Owner:[I;-1217574189,-1929301680,-1288476465,-1272613004],pickup:1b}
entity_seed arrow1 4
velocity arrow1 0 0 2.5
tick 4
dump entities advancements:A packets:A:damage_event,hurt_animation,add_entity,remove_entities
note an arrow without owner hits the third cow
spawn arrow2 minecraft:arrow 12.5 -59.5 2.5 0 0 {pickup:1b}
entity_seed arrow2 5
velocity arrow2 0 0 2.5
tick 4
dump entities advancements:A packets:A:damage_event,hurt_animation,add_entity,remove_entities
