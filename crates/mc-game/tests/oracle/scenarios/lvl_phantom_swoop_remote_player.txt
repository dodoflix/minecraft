# a phantom of the overworld that is swooping at its target when the target goes to the nether, to the same coordinates: the phantom loses its target
# (its targeting goal needs a line of sight, which never exists across levels), so it never bites and the player in the nether stays unhurt.
level void
command_env
end void
nether void
levels overworld,the_nether
preload 2
difficulty normal
entity_counter 100
level_seed 5
time 18000
fill -6 99 -6 12 99 12 minecraft:stone
in the_nether fill -6 99 -6 12 99 12 minecraft:netherrack
spawn_player Pl survival 0.5 100 0.5 0 0
mobspawn Ph minecraft:phantom 0.5 106 6.5 0 0 nofinalize {PersistenceRequired:1b}
entity_seed Ph 3
tick 2
dump packets:Pl:none
target Ph Pl
tick 14

player_loaded Pl
in the_nether dump packets:Pl:none
tick 60
dump mob:Ph
in the_nether dump player:Pl packets:Pl:damage_event,hurt_animation,set_health,entity_event,set_entity_motion,sound
tick 30
dump mob:Ph
