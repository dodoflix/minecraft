# Dispenser equipment on mobs that are not silent: LivingEntity.onEquipItem plays the equip sound with the entity's own sound source
# (hostile for zombie/skeleton, neutral for pig/horse/strider/wolf); the sound event is the item's equippable sound, overridden by
# Pig (PIG_SADDLE) and Strider (STRIDER_SADDLE). blk_disp_equip spawns every mob silent, so it only compares the player and armor stands.
level flat
preload 2
time 18000
# The targets are spawned first and ticked twice: an entity in its first tick plays no equip sound (LivingEntity.onEquipItem checks !firstTick).
spawn z1 minecraft:zombie 1.5 -60 0.5 0 0 {NoAI:1b,CanPickUpLoot:1b}
spawn s2 minecraft:skeleton 1.5 -60 4.5 0 0 {NoAI:1b,CanPickUpLoot:1b}
spawn p3 minecraft:pig 1.5 -60 8.5 0 0 {NoAI:1b}
spawn h4 minecraft:horse 1.5 -60 12.5 0 0 {NoAI:1b,Tame:1b}
spawn r5 minecraft:strider 1.5 -60 16.5 0 0 {NoAI:1b}
spawn w6 minecraft:wolf 1.5 -60 20.5 0 0 {NoAI:1b,Tame:1b}
tick 2
note c1 leather helmet on a non-silent zombie: sound category hostile
block 0 -60 0 minecraft:dispenser[facing=east,triggered=false]
be_nbt 0 -60 0 {Items:[{Slot:0b,id:"minecraft:leather_helmet",count:1}]}
level_seed 701
dispense 0 -60 0 c1
dump be:0,-60,0 entities events nbt:z1
note c2 iron helmet on a non-silent skeleton
block 0 -60 4 minecraft:dispenser[facing=east,triggered=false]
be_nbt 0 -60 4 {Items:[{Slot:0b,id:"minecraft:iron_helmet",count:1}]}
level_seed 702
dispense 0 -60 4 c2
dump be:0,-60,4 entities events nbt:s2
note c3 saddle on a non-silent pig: the pig saddle sound, category neutral
block 0 -60 8 minecraft:dispenser[facing=east,triggered=false]
be_nbt 0 -60 8 {Items:[{Slot:0b,id:"minecraft:saddle",count:1}]}
level_seed 703
dispense 0 -60 8 c3
dump be:0,-60,8 entities events nbt:p3
note c4 saddle on a non-silent tamed horse
block 0 -60 12 minecraft:dispenser[facing=east,triggered=false]
be_nbt 0 -60 12 {Items:[{Slot:0b,id:"minecraft:saddle",count:1}]}
level_seed 704
dispense 0 -60 12 c4
dump be:0,-60,12 entities events nbt:h4
note c5 saddle on a non-silent strider
block 0 -60 16 minecraft:dispenser[facing=east,triggered=false]
be_nbt 0 -60 16 {Items:[{Slot:0b,id:"minecraft:saddle",count:1}]}
level_seed 705
dispense 0 -60 16 c5
dump be:0,-60,16 entities events nbt:r5
