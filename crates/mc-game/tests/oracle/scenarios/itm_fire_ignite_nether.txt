# FlintAndSteelItem/FireChargeItem in the nether: an empty obsidian frame lights a nether portal, soul fire over soul sand.
dimension the_nether
preload 3
difficulty normal
watch events
# vanilla generates real nether terrain here: clear a room before the player joins
fill -6 64 -2 6 80 8 minecraft:air
fill -4 64 0 4 64 6 minecraft:stone
spawn_player Alice survival 0.5 65 0.5 0 0
tick 30
dump packets:Alice:none
level_seed 17
# Alice: survival
give Alice mainhand minecraft:flint_and_steel

# unlit campfire: flint and steel
fill -3 65 2 3 71 4 minecraft:air
