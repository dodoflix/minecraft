# DyeItem.interactLivingEntity on sheep: a live, unsheared sheep of another colour takes the dye (the stack shrinks, not for creative);
# the same colour, a sheared sheep and an item without a dye colour pass. Babies dye too.
level flat
preload 2
difficulty normal
spawn_player P survival 0.5 -60 0.5 0 0
spawn_player C creative 0.5 -60 8.5 0 0
give P mainhand minecraft:red_dye 8
give C mainhand minecraft:red_dye 8
mobspawn W minecraft:sheep 2.5 -60 0.5 90 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
mobspawn S minecraft:sheep 2.5 -60 2.5 90 0 nofinalize {NoAI:1b,PersistenceRequired:1b,Sheared:1b}
mobspawn B minecraft:sheep 2.5 -60 4.5 90 0 nofinalize {NoAI:1b,PersistenceRequired:1b,Age:-24000}
mobspawn R minecraft:sheep 2.5 -60 6.5 90 0 nofinalize {NoAI:1b,PersistenceRequired:1b,Color:14b}
mobspawn V minecraft:sheep 2.5 -60 8.5 90 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
tick 3
dump
note a white sheep takes the red dye: one dye used
interact P W main
tick 1
dump mob:W nbt:W inventory:P
note the sheep is red now: the same colour passes and the stack stays
interact P W main sneak
tick 1
dump mob:W nbt:W inventory:P

tick 1
dump mob:S nbt:S inventory:P

tick 1
dump mob:B nbt:B inventory:P

tick 1
dump mob:R nbt:R inventory:P
