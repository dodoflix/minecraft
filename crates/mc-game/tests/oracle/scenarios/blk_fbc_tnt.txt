# TntBlock: redstone ignition on place / neighbour update, flint and steel, fire charge, chain reaction, explosion
# (wasExploded primes neighbours with a short fuse), unstable tnt destroyed by a survival player, gamerule tnt_explodes.
level flat
preload 3
difficulty normal
watch events
spawn_player Alice survival 0.5 -60 0.5 0 0
give Alice mainhand minecraft:flint_and_steel
give Alice offhand minecraft:fire_charge 4
# chunk loading finishes asynchronously on the vanilla side: settle before anything draws level randomness; the join and
# give packets are only sent by the vanilla server, on its next tick
tick 30
dump packets:Alice:none
level_seed 99
# redstone ignition on neighbour update
block 5 -60 5 minecraft:tnt
block 5 -60 6 minecraft:redstone_block
dump blocks:4,-61,4,6,-59,7 entities
tick 1
dump blocks:4,-61,4,6,-59,7 entities packets:Alice
# placing a tnt next to a powered block primes it at once
block 8 -60 8 minecraft:redstone_block
block 8 -60 9 minecraft:tnt
dump entities
# flint and steel / fire charge
block 2 -60 0 minecraft:tnt
use_on Alice 2 -60 0 up
dump blocks:1,-61,-1,3,-59,1 entities inventory:Alice
select Alice 0
block 3 -60 0 minecraft:tnt
use_on Alice 3 -60 0 up off
dump blocks:2,-61,-1,4,-59,1 entities inventory:Alice
# a chain: tnt block row exploding; the source is a living entity that is not a player (the replay takes the source out of the
# level for the explosion, so a player source would miss the explosion packet)
spawn src minecraft:armor_stand 11.5 -60 20.5 0 0
fill 12 -60 0 16 -60 0 minecraft:tnt
explode 11 -60 0 3 none source=src
tick 1
dump blocks:10,-61,-1,17,-59,1 entities packets:Alice
tick 90
# remove_entities of an exploding tnt is sent after the explosion by the replay, vanilla sends it at the discard (known difference)
dump blocks:8,-61,-4,20,-58,4 entities events packets:Alice:!remove_entities
# unstable tnt dug by a survival player primes; creative does not
block 0 -60 10 minecraft:tnt[unstable=true]
dig Alice 0 -60 10
dump blocks:-1,-61,9,1,-59,11 entities packets:Alice:!remove_entities
# the game rule off: flint and steel only shows the overlay message
gamerule tnt_explodes false
block 0 -60 14 minecraft:tnt
use_on Alice 0 -60 14 up
dump blocks:-1,-61,13,1,-59,15 entities packets:Alice:!remove_entities
