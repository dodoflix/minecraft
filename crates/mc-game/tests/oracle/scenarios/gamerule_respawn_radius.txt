# respawn_radius: the search for a player's first spawn position starts at the suggestion and tries the columns of the square of that
# radius around it in a random order (radius 0: the suggested column only). Eight searches per radius; within = all inside the square.
command_env
level flat
preload 3
level_seed 3
entity_counter 50
spawn_player A survival 0.5 -60 0.5 0 0
tick 1
find_spawn 0 -60 0 8
cmd gamerule respawn_radius 0
find_spawn 0 -60 0 8
cmd gamerule respawn_radius 3
find_spawn 0 -60 0 8
cmd gamerule respawn_radius 1
find_spawn 20 -60 20 8
