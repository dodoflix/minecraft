# Counting slots reads no item: `execute if slots` leaves a pending loot table alone, a `filtered` source over the container reads
# every slot (and unpacks), and `/loot replace entity` writes a chest vehicle slot through setChestVehicleItem (unpack first). Blocks and
# vehicles with a pending table; each case has its own targets.
datapack loot_cmds
command_env
level flat
preload 3
level_seed 2401
entity_counter 2400
spawn_player A survival 0.5 -60 40.5 0 0
op A
gamerule send_command_feedback true
tick 1
note chest: execute if slots keeps the table, a filtered source unpacks
cmd setblock 2 -60 2 minecraft:chest{LootTable:"minecraft:chests/simple_dungeon",LootTableSeed:5L}
cmd execute if slots block 2 -60 2 container.*
cmd data get block 2 -60 2 LootTable
cmd execute if items block 2 -60 2 {type:"minecraft:filtered",slot_source:{type:"minecraft:slot_range",source:"container",slots:"container.*"},item_filter:{items:"minecraft:bone"}} *
cmd data get block 2 -60 2 LootTable
note barrel: execute if slots keeps the table, a filtered source unpacks
cmd setblock 2 -60 6 minecraft:barrel[facing=up]{LootTable:"minecraft:chests/simple_dungeon",LootTableSeed:6L}
cmd execute if slots block 2 -60 6 container.*
cmd data get block 2 -60 6 LootTable
cmd execute if items block 2 -60 6 {type:"minecraft:filtered",slot_source:{type:"minecraft:slot_range",source:"container",slots:"container.*"},item_filter:{items:"minecraft:bone"}} *
cmd data get block 2 -60 6 LootTable
