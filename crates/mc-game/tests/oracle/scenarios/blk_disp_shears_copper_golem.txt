# ShearsDispenseItemBehavior on a copper golem: the antenna item (a poppy) is shearable and drops 1.5 above the feet with the
# shear sound; a golem without the antenna item or with another item in the antenna slot is ignored (failure click).

preload 4

spawn g1 minecraft:copper_golem 1.5 -60 0.5 0 0 {NoAI:1b,Silent:1b,equipment:{saddle:{id:"minecraft:poppy",count:1}}}
