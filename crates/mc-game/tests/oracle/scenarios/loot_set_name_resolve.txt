# set_name and set_lore resolve selector, score and nbt text with the loot context entity as the source (ComponentUtils.resolve with
# entity.createCommandSourceStackForNameResolution(level) at game master): /item modify on a player hotbar slot, a container and an
# entity, and /loot give, replace and spawn-less tables with an inline table. Without an entity the text stays as written.
command_env
level flat
preload 2
level_seed 3
entity_counter 50
spawn_player Op survival 0.5 -60 0.5 0 0
spawn_player Plain survival 4.5 -60 0.5 0 0

cmd_quiet scoreboard objectives add o dummy
cmd_quiet scoreboard players set Op o 7
cmd_quiet scoreboard players set Plain o 9
cmd_quiet scoreboard players set Bob o 3
cmd_quiet data merge storage t:s {str:"hi",num:5,list:[1,2,3],comp:{text:"stored",color:"gold"},sel:{selector:"@s"}}
cmd_quiet summon minecraft:pig 3 -60 3 {UUID:[I;1,2,3,4],CustomName:"Bob",NoAI:1b,Tags:["t1"]}
cmd_quiet setblock 0 -60 4 minecraft:oak_sign[rotation=0]
cmd_quiet data merge block 0 -60 4 {front_text:{messages:["one",{text:"two",color:"red"},"",""]}}
cmd_quiet setblock 2 -60 4 minecraft:chest
cmd_quiet item replace entity Op hotbar.0 with minecraft:stick 3
