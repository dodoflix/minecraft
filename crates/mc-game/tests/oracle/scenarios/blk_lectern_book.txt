# Lectern: placing a book (sound, has_book, comparator, a lamp under it), opening the menu, turning pages (powered pulse of 2 ticks,
# level event 1043, comparator signal by page, clamping), jumping to a page, taking the book, and a written book.
command_env
level flat
preload 2
spawn_player A survival 0.5 -60 3.5 180 0
spawn_player W survival 8.5 -60 3.5 180 0
block 0 -61 0 minecraft:redstone_lamp[lit=false]
block 0 -60 0 minecraft:lectern[facing=south,has_book=false,powered=false]
block 1 -60 0 minecraft:comparator[facing=east,mode=compare,powered=false]
block 2 -60 0 minecraft:redstone_lamp[lit=false]
block 8 -61 0 minecraft:redstone_lamp[lit=false]
block 8 -60 0 minecraft:lectern[facing=south,has_book=false,powered=false]
tick 2
dump blocks:-1,-61,-1,3,-59,1 events packets:A:none packets:W:none
note an empty lectern does nothing with an empty hand or an unrelated item
use_on A 0 -60 0 up main
give A mainhand minecraft:apple
use_on A 0 -60 0 up main
dump events packets:A:open_screen,container_close blocks:0,-60,0,0,-60,0
note a writable book is placed
give A mainhand minecraft:writable_book[writable_book_content={pages:["a","b","c","d","e"]}]
use_on A 0 -60 0 up main
dump events blocks:-1,-61,-1,3,-59,1 be:0,-60,0 inventory:A
tick 6
dump events blocks:-1,-61,-1,3,-59,1
note a second book is not accepted, the empty hand opens the menu
give A mainhand minecraft:writable_book[writable_book_content={pages:["x"]}]
use_on A 0 -60 0 up main
dump events packets:A:open_screen,container_close inventory:A
give A mainhand minecraft:air
use_on A 0 -60 0 up main
dump events packets:A:open_screen,container_close,container_set_content,container_set_data blocks:0,-60,0,0,-60,0

menu_button A 2

dump events blocks:-1,-61,-1,3,-59,1 ticks
tick 1

tick 6
dump events blocks:-1,-61,-1,3,-59,1 ticks packets:A:container_set_data
menu_button A 2

menu_button A 2
tick 4
dump events blocks:-1,-61,-1,3,-59,1 be:0,-60,0
