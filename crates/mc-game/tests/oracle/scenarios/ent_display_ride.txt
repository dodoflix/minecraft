# Display entities as passengers: a display rides a boat and leaves it when the boat is removed (its own tick is only that
# check), is carried along while it rides, and is a plain passenger in the passenger packets; /item replace entity writes
# the stack of an item display through its slot 0 (contents), and no other slot or entity takes it.
command_env
level flat
preload 2
level_seed 5
entity_counter 1000
spawn_player A survival 0.5 -60 0.5 0 0
tick 2
dump packets:A:none
spawn V minecraft:oak_boat 0.5 -60 5.5 0 0
spawn D1 minecraft:text_display 0.5 -60 5.5 0 0 {text:"rider"}
spawn D2 minecraft:item_display 0.5 -60 5.5 0 0 {item:{id:"minecraft:stone",count:1}}
ride D1 V
ride D2 V
tick 3
dump entities entity:D1 entity:D2 nbt:D1 packets:A:add_entity,set_entity_data,set_passengers
note the boat moves and takes its passengers along
tp V 4.5 -60 5.5
tick 3
dump entity:V entity:D1 entity:D2 packets:A:set_passengers,teleport_entity,move_entity_pos,entity_position_sync
note the boat is removed: the displays stop riding on their next tick
kill V
tick 3
dump entities entity:D1 entity:D2 packets:A:set_passengers,remove_entities,set_entity_data
note /item replace entity on an item display: slot 0 (contents) takes the item, others fail, an empty replace clears it
cmd item replace entity @e[type=minecraft:item_display,limit=1] contents with minecraft:emerald 5
cmd item replace entity @e[type=minecraft:item_display,limit=1] weapon.mainhand with minecraft:stick
cmd item replace entity @e[type=minecraft:text_display,limit=1] contents with minecraft:stick
cmd execute if items entity @e[type=minecraft:item_display,limit=1] contents minecraft:emerald run say contents is an emerald
cmd data get entity @e[type=minecraft:item_display,limit=1] item
tick 2
dump nbt:D2 nbt:D1 packets:A:set_entity_data
cmd item replace entity @e[type=minecraft:item_display,limit=1] contents with minecraft:air
cmd item modify entity @e[type=minecraft:item_display,limit=1] contents {type:"minecraft:set_count",count:4}
cmd item replace entity @e[type=minecraft:item_display,limit=1] contents with minecraft:stone 64
cmd item modify entity @e[type=minecraft:item_display,limit=1] contents {type:"minecraft:set_name",name:{text:"shown"}}
tick 2
dump nbt:D2 packets:A:set_entity_data
note displays do not press plates or trip wires and are not moved by a piston
block 4 -61 7 minecraft:stone_pressure_plate
block 4 -60 7 minecraft:air
spawn D3 minecraft:block_display 4.5 -60 7.5 0 0 {block_state:"minecraft:stone"}
spawn D4 minecraft:text_display 4.5 -60 7.5 0 0 {text:"plate"}
tick 3
dump blocks:4,-61,7,4,-61,7 entities
block 7 -60 9 minecraft:piston[facing=east]
block 8 -60 9 minecraft:air
spawn D5 minecraft:block_display 8.5 -60 9.5 0 0 {block_state:"minecraft:stone"}
block 6 -60 9 minecraft:redstone_block
tick 4
dump blocks:7,-60,9,8,-60,9 entity:D5 entities packets:A:move_entity_pos,teleport_entity,entity_position_sync
