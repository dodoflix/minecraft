# A player who leaves while riding: `PlayerList.remove` saves the player with its root vehicle (and the passengers of that vehicle) and takes the mount out of
# the level with it; joining again with the saved data spawns the vehicle and its passengers and puts the player back on it. A horse and a boat with a second
# passenger are tried.
level void
command_env
preload 2
difficulty normal
entity_counter 800
level_seed 19
fill -8 -61 -8 40 -61 8 minecraft:stone
spawn_player Alice survival 0.5 -60 0.5 0 0
mobspawn Horse minecraft:horse 3.5 -60 0.5 0 0 nofinalize {Tame:1b,PersistenceRequired:1b,SaddleItem:{id:"minecraft:saddle",count:1},Variant:2}
spawn Boat minecraft:oak_boat 20.5 -60 0.5 90 0
mobspawn Pig minecraft:pig 20.5 -60 0.5 0 0 nofinalize {NoAI:1b,PersistenceRequired:1b}
tick 2
ride Alice Horse
accept_teleport Alice
tick 3
dump players entities nbt:Alice
note Alice leaves on her horse: the horse goes with her
leave Alice
tick 2
dump players entities playerfile:Alice
note Alice joins again with her saved data: the horse is back under her
rejoin Alice
tick 3
dump players entities nbt:Alice
note the same in a boat with a second passenger
ride Pig Boat
tp Alice 20.5 -60 0.5
ride Alice Boat
accept_teleport Alice
tick 3
leave Alice
tick 2
dump players entities playerfile:Alice
rejoin Alice
tick 3
dump players entities nbt:Alice
