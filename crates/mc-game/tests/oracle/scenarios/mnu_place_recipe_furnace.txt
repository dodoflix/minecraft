# The recipe book placing a recipe into the furnace, smoker and blast furnace (ServerboundPlaceRecipe): the ingredient moves from the
# inventory into the input slot (one container_set_slot per change), the input and result slots are cleared first, a recipe that cannot
# be made clears them and gets the ghost recipe packet, and the cases the listener ignores. Slots: 0 input, 1 fuel, 2 result,
# 3-29 inventory, 30-38 hotbar. Player A uses the furnace, B the smoker, C the blast furnace, D is a spectator.
command_env
level flat
preload 4
spawn_player A survival 4.5 -60 6.5 0 0
spawn_player B survival 10.5 -60 6.5 0 0
spawn_player C survival 16.5 -60 6.5 0 0
spawn_player D spectator 22.5 -60 6.5 0 0
block 3 -60 4 minecraft:furnace[facing=south,lit=false]
block 10 -60 4 minecraft:smoker[facing=south,lit=false]
block 16 -60 4 minecraft:blast_furnace[facing=south,lit=false]
block 22 -60 4 minecraft:furnace[facing=south,lit=false]
give A inv9 minecraft:raw_iron 64
give A inv10 minecraft:raw_iron 20
give A inv11 minecraft:sand 10
give A inv12 minecraft:beef 5
give A inv13 minecraft:coal 3
give B inv9 minecraft:beef 40
give B inv10 minecraft:potato 5
give C inv9 minecraft:raw_gold 30
give C inv10 minecraft:iron_sword 1
give D inv9 minecraft:raw_iron 4

cmd recipe give A minecraft:iron_ingot_from_smelting_raw_iron
cmd recipe give A minecraft:glass

cmd recipe give C minecraft:gold_ingot_from_blasting_raw_gold
cmd recipe give C minecraft:iron_nugget_from_blasting
cmd recipe give D minecraft:iron_ingot_from_smelting_raw_iron
use_on A 3 -60 4 up main
use_on B 10 -60 4 up main
use_on C 16 -60 4 up main
use_on D 22 -60 4 up main
dump menu:A packets:A:container_set_slot,container_set_content,set_cursor_item,place_ghost_recipe recipes:A packets:B:none packets:C:none packets:D:none
