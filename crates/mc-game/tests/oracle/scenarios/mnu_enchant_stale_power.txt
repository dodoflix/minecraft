# The enchanting table offers are computed when a slot changes: bookshelves removed afterwards do not change the offers or what a
# button gives; the next slot change recomputes them.
command_env
level flat
preload 4
spawn_player A survival 3.5 -60 7.5 0 0
block 3 -60 4 minecraft:enchanting_table
fill 1 -60 2 5 -59 2 minecraft:bookshelf
fill 1 -60 6 5 -59 6 minecraft:bookshelf
fill 1 -60 3 1 -59 5 minecraft:bookshelf
fill 5 -60 3 5 -59 5 minecraft:bookshelf
set A xp_level 40
set A enchant_seed 999
entity_seed A 5
give A hotbar0 minecraft:diamond_sword
give A hotbar1 minecraft:lapis_lazuli 10
give A hotbar2 minecraft:book 3
tick 2
use_on A 3 -60 4 up main
click A 29 0 PICKUP
click A 0 0 PICKUP
click A 30 0 PICKUP
click A 1 0 PICKUP
dump menu:A packets:A:open_screen,container_set_slot,container_set_content,container_set_data,set_cursor_item
note break all the shelves, the offers stay until the next slot change
fill 1 -60 2 5 -59 6 minecraft:air
tick 3
dump menu:A packets:A:container_set_slot,container_set_content,container_set_data,set_cursor_item
note a different item: recomputed without shelves
click A 0 0 PICKUP
click A 31 0 PICKUP
click A 0 0 PICKUP
dump menu:A packets:A:container_set_slot,container_set_content,container_set_data,set_cursor_item
note a button of a slot that no longer offers anything
menu_button A 2
menu_button A 0
dump menu:A player:A stats:A packets:A:container_set_slot,container_set_content,container_set_data,set_cursor_item,set_experience
