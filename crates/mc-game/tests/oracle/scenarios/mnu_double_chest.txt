# A real double chest (right half at x=2, left half at x=3, facing south): the 9x6 menu with the double chest title, both halves
# in one menu, clicks across the halves, the stat and a spectator opening it; breaking one half under the open menu closes it.
command_env
level flat
preload 4
spawn_player A survival 2.5 -60 3.5 180 0
spawn_player S spectator 4.5 -60 3.5 180 0
block 2 -60 0 minecraft:chest[facing=south,type=right,waterlogged=false]
block 3 -60 0 minecraft:chest[facing=south,type=left,waterlogged=false]
be_nbt 2 -60 0 {Items:[{Slot:0b,id:"minecraft:stone",count:30},{Slot:26b,id:"minecraft:dirt",count:5}]}
be_nbt 3 -60 0 {Items:[{Slot:0b,id:"minecraft:diamond",count:7},{Slot:26b,id:"minecraft:gold_ingot",count:9}]}
give A hotbar0 minecraft:apple 10
tick 2
dump blocks:2,-60,0,3,-60,0 packets:A:none packets:S:none stats:A
level_seed 11
use_on A 2 -60 0 south main
dump menu:A stats:A packets:A:open_screen,container_set_content,container_set_data events
click A 0 0 PICKUP
click A 30 0 PICKUP
click A 53 0 QUICK_MOVE
click A 27 0 QUICK_MOVE
click A 54 0 PICKUP_ALL
dump menu:A inventory:A packets:A:container_set_slot,container_set_content,set_cursor_item be:2,-60,0 be:3,-60,0
use_on S 3 -60 0 south main
dump menu:S packets:S:open_screen,container_set_content,container_set_data
note breaking one half closes the menu of the player using the double chest
destroy 3 -60 0
tick 1
dump menu:A menu:S packets:A:container_close,container_set_content packets:S:container_close entities
