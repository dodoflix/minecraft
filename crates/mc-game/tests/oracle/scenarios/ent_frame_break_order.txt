# The packets of an item frame that holds a map going away, in order: a hit knocks the map out (the dropped map appears, the
# frame's data changes), a second hit breaks the frame (the dropped frame, the sound, remove_entities), a frame that loses its
# support block, and a creative player's hit. The map data forgets the frame in each case; the observer holds the map.
command_env
level flat
preload 3
level_seed 5
entity_counter 980
spawn_player Hit survival 6.5 -60 2.5 90 0
spawn_player Obs survival 6.5 -60 12.5 90 0
spawn_player Cre creative 6.5 -60 22.5 90 0
block 3 -60 0 minecraft:stone
block 3 -60 10 minecraft:stone
block 3 -60 20 minecraft:stone
tick 2
dump packets:Hit:none packets:Obs:none packets:Cre:none
give Obs hotbar0 minecraft:map 1
select Obs 0
use Obs main
select Obs 1
tick 1
dump inventory:Obs packets:Obs:none
note a frame with the map hit twice: the map drops out, then the frame breaks
spawn f1 minecraft:item_frame 4.5 -59.5 0.5 0 0 {Facing:5b,Item:{id:"minecraft:filled_map",count:1,components:{"minecraft:map_id":0}}}
entity_seed f1 51
tick 12
dump packets:Hit:none packets:Obs:map_item_data
level_seed 6
attack Hit f1
dump xentities packets:Hit:add_entity,remove_entities,set_entity_data,sound packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
tick 12
dump xentities packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
level_seed 7
attack Hit f1 second
dump xentities packets:Hit:add_entity,remove_entities,set_entity_data,sound packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
tick 12
dump xentities packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
note a frame with the map loses its support block
spawn f2 minecraft:item_frame 4.5 -59.5 10.5 0 0 {Facing:5b,Item:{id:"minecraft:filled_map",count:1,components:{"minecraft:map_id":0}}}
entity_seed f2 52
tick 12
dump packets:Obs:map_item_data
level_seed 8
block 3 -60 10 minecraft:air
tick 105
dump xentities packets:Obs:remove_entities,add_entity,sound,set_entity_data,map_item_data
note a creative player breaks a frame holding a map: no drops
spawn f3 minecraft:item_frame 4.5 -59.5 20.5 0 0 {Facing:5b,Item:{id:"minecraft:filled_map",count:1,components:{"minecraft:map_id":0}}}
entity_seed f3 53
tick 12
dump packets:Obs:map_item_data packets:Cre:none
attack Cre f3
dump xentities packets:Cre:add_entity,remove_entities,set_entity_data,sound packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
attack Cre f3 second
dump xentities packets:Cre:add_entity,remove_entities,set_entity_data,sound packets:Obs:add_entity,remove_entities,set_entity_data,sound,map_item_data
tick 12
dump xentities packets:Obs:map_item_data
